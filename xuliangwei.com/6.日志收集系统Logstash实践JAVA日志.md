    file {
        type => "access_es"
        path => "/var/log/elasticsearch/xuliangwei.log"
        codec => multiline {
            pattern => "^\["
            negate => true
            what => "previous"
            }
        }
}

        host => "192.168.90.204"
        port => "6379"
        db => "6"
        data_type => "list"
        key => "access_es"
        }
}
